// TASK 5 : THREE MONGODB AGGREGATION PIPELINES USING $lookup
// ============================================================

//in - Pipeline 1: Combine loan documents with customer details
print("\n--- PIPELINE 1: LOANS WITH CUSTOMER DETAILS ---");

db.loans.aggregate([
    {
        $lookup: {
            from: "customers",
            localField: "customerId",
            foreignField: "customerId",
            as: "customer"
        }
    },
    {
        $unwind: "$customer"
    },
    {
        $project: {
            _id: 0,
            loanId: 1,
            loanAmount: 1,
            outstandingBalance: 1,
            loanStatus: 1,
            customerId: 1,
            customerName: "$customer.customerName",
            customerCity: "$customer.city"
        }
    },
    { $sort: { loanId: 1 } }
]);


// ============================================================
// PIPELINE 2 : LOANS WITH ALL REPAYMENTS
// ============================================================

//in - Combine loan documents with all repayment records
print("\n--- PIPELINE 2: LOANS WITH ALL REPAYMENTS ---");

db.loans.aggregate([
    {
        $lookup: {
            from: "repayments",
            localField: "loanId",
            foreignField: "loanId",
            as: "repaymentHistory"
        }
    },
    {
        $project: {
            _id: 0,
            loanId: 1,
            customerId: 1,
            loanAmount: 1,
            repaymentHistory: 1
        }
    },
    { $sort: { loanId: 1 } }
]);


// ============================================================
// PIPELINE 3 : LOANS WITH CUSTOMER AND LAST REPAYMENT
// ============================================================

//in - Combine each loan with customer details and its LAST repayment
//The lookup pipeline sorts repayments newest-first and keeps only one record.
print("\n--- PIPELINE 3: LOANS WITH CUSTOMER AND LAST REPAYMENT ---");

db.loans.aggregate([
    {
        $lookup: {
            from: "customers",
            localField: "customerId",
            foreignField: "customerId",
            as: "customer"
        }
    },
    {
        $unwind: {
            path: "$customer",
            preserveNullAndEmptyArrays: true
        }
    },
    {
        $lookup: {
            from: "repayments",
            let: { currentLoanId: "$loanId" },
            pipeline: [
                {
                    $match: {
                        $expr: {
                            $eq: ["$loanId", "$$currentLoanId"]
                        }
                    }
                },
                { $sort: { repaymentDate: -1, repaymentId: -1 } },
                { $limit: 1 }
            ],
            as: "lastRepayment"
        }
    },
    {
        $unwind: {
            path: "$lastRepayment",
            preserveNullAndEmptyArrays: true
        }
    },
    {
        $project: {
            _id: 0,
            loanId: 1,
            customerName: "$customer.customerName",
            loanAmount: 1,
            outstandingBalance: 1,
            lastRepayment: {
                repaymentId: "$lastRepayment.repaymentId",
                repaymentDate: "$lastRepayment.repaymentDate",
                amount: "$lastRepayment.amount",
                status: "$lastRepayment.status"
            }
        }
    },
    { $sort: { loanId: 1 } }
]);
